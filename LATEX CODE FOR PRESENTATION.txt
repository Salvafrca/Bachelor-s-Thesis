\begin{frame}{Primera Formulación (1925-1932)}

 \begin{itemize}
     
     \item \textbf{Búsqueda de la formalización}
     \begin{itemize}
         \item Base Matricial (W. Heisenberg)
         \item Base Funcional (E. Schrördinger)
         \item Isomorfismo (J. von Neumann) 
     \end{itemize}
      \pause

      
     \item  \textbf{Estructuras de Estados }
     \begin{itemize}
         \item Estados: $\mathscr{H}$
         \item Cuerpo es $\mathbb{C}$
     \end{itemize}
     \pause

    \item \textbf{Estrucutra para los Operadores} 
        \begin{itemize}        
         \item Operadores: $\mathcal{B}(\mathscr{H})$
         \item Observables:
         En particular $A\in \mathcal{B}(\mathscr{H}), A = A^\dagger$
         \item Asociatividad $(AB)C = A(BC)$
         \item No Conmutatividad $AB \neq BA$
         \item $\dim(\mathcal{H}) = \infty$
     \end{itemize}
     \pause

    \item \textbf{Limitaciones o fallos}
     \begin{itemize}
         \item Sin justificación física para $\mathbb{C}$
         \item Sin justificación física parala Asociatividad 
     \end{itemize}

 \end{itemize}

\end{frame}






\begin{frame}{Axiomatización del Modelo (1934)}

 \begin{itemize}

    \item \textbf{Motivación de la nueva formulación}
    \begin{itemize}
        \item Solventar las carencias del $\mathcal{B}(\mathscr{H})$
        \item Búsqueda de una \textit{axiomatización} y de una \textit{generalización}
        \item Poniendo el foco en los \textit{operadores} y no en los \textit{estados}
    \end{itemize}
    \pause


    \item \textbf{Se busca}
    \begin{itemize}
        \item Estructura para los Observables
        \item Pleno significado físico
    \end{itemize}

    \item \textbf{Punto de partida}\\
    Sea $\mathcal{A}$ el espacio de observables y $A, B \in \mathcal{A}$:
    \begin{itemize}
        \item Suma: $A + B \in \mathcal{A}$
        \item Potencias: $A^n \in \mathcal{A}$
        \item Producto por escalar:$\lambda A \in \mathcal{A}$
    \end{itemize}

    
\end{itemize}

\end{frame}





\begin{frame}{\textit{Álgebras de Jordan}}
  \begin{columns}[c]
    \begin{column}{0.6\textwidth}
      \begin{itemize}
        \item Se trata de un álgebra $\mathcal{A}$ sobre $\mathbb{R}$. Sean $A, B \in \mathcal{A}$ el producto $\circ$ cumple:
          \begin{itemize}
            \item Conmutatividad: $A \circ B = B \circ A$
            \item Identidad de Jordan: $(A \circ B) \circ (A \circ A) = A \circ (B \circ (A \circ A))$
            \item No necesariamente asociativo
            \item Obtención a partir de un ágebra asociativa
          \end{itemize}
      \end{itemize}
    \end{column}
    \begin{column}{0.4\textwidth}
      \begin{figure}
        \centering
        \includegraphics[width=\columnwidth, height=0.65\textheight, keepaspectratio]{JORDAN.png}
        
      \end{figure}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{La formulación para dimensión infinita (1940-1950)}
\begin{itemize}
    \item Estados: funcionales lineales sobre el álgebra que toman valores y que verifican:
    \begin{itemize}
        \item $\omega(U^2)\geq0$
        \item $\omega(I)=1$
    \end{itemize}
    \pause

    \item Estado Puro
    \pause
    
    \item Valor esperado para un observable $U$ es $\omega(U)$
    \pause

    \item Completitud de estados puros 
    \pause

    \item Resultado fundamental para la distinción de obersvables
    \pause
    \item Álgebra para esta formulación $C^*-álgebra$
    \pause
\end{itemize}
    
\end{frame}









\begin{frame}{\textit{C$^*$-álgebra}}

\begin{itemize}
Definidas por I. Gelfand y M. Naimark

    \item Se trata de un álgebra $\mathcal{A}$ sobre el cuerpo $\mathbb{C}$
    \pause

    \item Adición de una norma $||\cdot||$ (completa)
    \pause

    \item Involución \qquad $*: A \to A$ tal que $x \mapsto x^*$
    \pause

    \item Propiedades a cumplir
    \begin{itemize}
        \item $||xy|| \le ||x|| ||y||$
        \item $||x^* x|| = ||x||^2$
    \end{itemize}
    
    \item Existe la posibilidad de obtener una $C^*$-álgebra real.
\end{itemize}

\begin{frame}{JB-álgebra}

    \begin{itemize}
        Definidas por Alfsen y Schultz en colaboración con E. Størmer
        \item Se trata de un álgebra $\mathcal{A}$ sobre $\mathbb{R}$
        \item Añadiendo una norma $||\cdot||$ \textit{completa}
        \pause
        
        \item Propiedades:
        \begin{itemize}
            \item Submultiplicatividad de la norma $\|a \circ b\| \leq \|a\|\|b\|$
            \item Identidad isométrica $\|a^2\| = \|a\|^2$
            \item Condición de positividad $\|a^2\| \leq \|a^2 + b^2\|$
        \end{itemize}
    \end{itemize}

\begin{frame}{Conclusiones}
    \begin{itemize}
        \item \textbf{Enfoque del trabajo:} 
        \begin{itemize}
            \item Conectar la profunda abstracción matemática de la mecánica cuántica con la intuición física.
            \item Un esfuerzo enriquecedor para dimensionar la evolución histórica de estos modelos.
        \end{itemize}
        
        \vspace{0.3cm} % Espacio extra para separar los bloques
  
        
        \item \textbf{P1: ¿Estructura algebraica subyacente para los observables?}
        \begin{itemize}
            \item La estructura definitiva son las \textbf{JB-álgebras} (sobre $\mathbb{R}$ y en dimensión infinita).
            \item Elimina artificios (como la asociatividad impuesta y el uso de $\mathbb{C}$), logrando un modelo fiel a la realidad del laboratorio.
        \end{itemize}
        
        \vspace{0.3cm}


        \item \textbf{P2: ¿Geometría en el espacio de estados?}
        \begin{itemize}
            \item Sí, gracias a la \textbf{axiomatización del Espacio de Estados} (Alfsen y Shultz).
            \item Se deduce la estructura de los observables directamente a partir de la geometría de los estados puros.
            \item Se le devuelve el protagonismo a la Física (lo que medimos) frente a la Matemática pura.
        \end{itemize}
    \end{itemize}
\end{frame}
